switch_destroyonreset(colaAcabado, 0);
switch_destroyonreset(acabado, 0);

colaAcabado.name = "Cola_Acabado";
acabado.name = "Acabado";
productoTerminado.name = "Producto_Terminado";

//
// OnExit se evalua antes de confirmar la salida actual. Por eso output=419
// identifica la salida de la pieza 420. En ese instante se coloca la proxima
// llegada fuera del horizonte y se destruye cualquier temporizador ya creado.
// OnReset restaura el intervalo para que Reiniciar + Ejecutar reproduzca el lote.

    "Object current = ownerobject(c);\n"
    + "current.setProperty(\"InterArrivalTime\", 0.06);\n";
materiaPrima.setProperty("Triggers.OnReset", sourceResetCode);

string sourceExitCode =
    "Object current = ownerobject(c);\n"
    + "Object item = param(1);\n"
    + "int port = param(2);\n"
    + "if (current.stats.output.value >= 419) {\n"
    + "    current.setProperty(\"InterArrivalTime\", hours(1000));\n"
    + "    destroyeventsofobject(current);\n"
    + "}\n";
materiaPrima.setProperty("Triggers.OnExit", sourceExitCode);
